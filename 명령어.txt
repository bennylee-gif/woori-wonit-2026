echo "# woori-wonit-2026" >> README.md
// echo 뒤에 오는 텍스트를 출력하는 명령어
// >> 출력을 파일 뒤에 덧붙여 저장
// README.md 라는 파일에 # woori-wonit-2026 이라는 큰 제목 글자를 써서 파일 새로 생성(추가)

git init
// init initialize, 현재 폴더를 git 저장소로 시작하겠다
// 지금 현재 폴더를 git이 버전 관리하는 폴더로 지정

git add README.md
// add 파일의 변경사항을 저장 대기 공간에 올리는 명령어
// README.md 파일을 다음 커밋(저장)에 포함할 대상으로 장바구니에 담기

git commit -m "first commit"
// commit 확정적으로 기록
// -m 커밋 메시지
// first commit 이라는 설명 태그를 붙여 내 컴퓨터에 확정적 저장

git branch -M main
// branch 작업 가지 관리
// 생성된 작업 줄기(가지) 이름을 main으로 변경

git remote add origin https://github.com/bennylee-gif/woori-wonit-2026.git
// remote add 원격 저장소의 주소 등록
// 원격 저장소를 저장해서 그 주소를 이제부터 origin으로 부르겠다

git push -u origin main
// push 내 컴퓨터의 기록을 웹으로 올리는 명령어
// -u 브랜치의 연결관계 기억
// 내 컴퓨터의 main 브랜치 기록을 원격 저장소의 브랜치로 업로드하고, 앞으로는 바로 연결될 수 있게 단축설정

git config --global user.email "bennylee@sookmyung.ac.kr"
git config --global user.name "Benny"

######## 새로 내용을 수정했을 때
git add README.md
git commit -m "second commit"
git push -u origin main
